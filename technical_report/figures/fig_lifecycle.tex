\begin{figure}[t]
\centering
\begin{schematicpanel}{Figure 2. Schematic: one mission's bounded engineer/reviewer loop with context and stall guards (control flow, not measured data)}
\centering
\begin{tikzpicture}[
  font=\footnotesize,
  node distance=0.9cm,
  stepbox/.style={draw=argusindigo, fill=calloutfill, rounded corners=2pt,
               align=center, minimum height=0.95cm, minimum width=2.7cm,
               text=graphite, line width=0.6pt},
  guard/.style={draw=panelline, fill=panelfill, rounded corners=2pt,
               align=center, minimum height=0.9cm, minimum width=2.9cm,
               text=graphitesoft, line width=0.6pt},
  flow/.style={-{Stealth[length=2mm]}, draw=graphitesoft, line width=0.5pt}
]
  \node[stepbox] (prompt) {Build round\\prompt};
  \node[stepbox, right=1.1cm of prompt] (run) {Engineer\\runs round $k$};
  \node[stepbox, right=1.1cm of run] (review) {Reviewer\\evaluates};
  \node[stepbox, right=1.1cm of review] (verdict) {\code{done} /\\\code{continue} /\\\code{blocked}};

  \draw[flow] (prompt) -- (run);
  \draw[flow] (run) -- (review);
  \draw[flow] (review) -- (verdict);
  \draw[flow] (verdict.south) .. controls +(0,-0.9) and +(0,-0.9) ..
     node[midway, below, text=argusindigo]{\scriptsize \code{continue}: reseed next round from checkpoint}
     (prompt.south);

  \node[guard, below=1.7cm of prompt] (ckpt) {Curated checkpoint\\{\scriptsize goal / done / tried /}\\{\scriptsize blocker / next}};
  \node[guard, below=1.7cm of run] (roll) {Session roll\\{\scriptsize 3 rounds \emph{or} 1.5M}\\{\scriptsize input tokens}};
  \node[guard, below=1.7cm of review] (stall) {Stall guard\\{\scriptsize 2 nondecision rounds,}\\{\scriptsize 1{,}800\,s round budget}};
  \node[guard, below=1.7cm of verdict] (audit) {Audit trail\\{\scriptsize evidence, verdict,}\\{\scriptsize skill update}};

  \draw[flow, dashed, draw=indigosoft] (review.south) -- (ckpt.north);
  \draw[flow, dashed, draw=indigosoft] (ckpt.east) -- (roll.west);
  \draw[flow, dashed, draw=indigosoft] (roll.east) -- (stall.west);
  \draw[flow, dashed, draw=indigosoft] (stall.east) -- (audit.west);
\end{tikzpicture}
\end{schematicpanel}
\vspace{-0.6em}
\end{figure}
